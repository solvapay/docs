Use this guide when you want to monetize an MCP server and keep implementation control in your own codebase.

<Note>
If you want SolvaPay to host auth, checkout, and paywall enforcement with no
code changes, use [Managed MCP](/guides/monetize-mcp-server-no-code) instead.
</Note>

## Who this is for

- Teams building custom MCP server behavior
- Teams that need custom auth and entitlement logic
- Teams that want SDK-level control instead of Managed MCP

## What you will achieve

By the end, your MCP server uses the SolvaPay SDK to enforce purchases and manage monetized tool access.

## What the change looks like

You build the server with SolvaPay's MCP server factory and register each paid tool with it.
Your tool logic stays the same. Before each call, SolvaPay checks the customer's plan: the call
either runs and is billed, or returns the paywall with a checkout link. The factory also adds
the `account` and `activate_plan` tools and the checkout widget.

```typescript TypeScript
import { z } from 'zod'
import { createSolvaPayMcpServer } from '@solvapay/mcp'

const server = createSolvaPayMcpServer({
  solvaPay,
  productRef: process.env.SOLVAPAY_PRODUCT_REF,
  resourceUri: 'ui://my-app/mcp-app.html',
  htmlPath: path.join(import.meta.dirname, 'mcp-app.html'),
  publicBaseUrl: 'https://my-mcp.example.com',
  additionalTools: ({ registerPayable }) => {
    registerPayable('search_knowledge', {
      description: 'Search the knowledge base.',
      schema: { query: z.string().min(1) },
      annotations: { readOnlyHint: true, idempotentHint: true },
      handler: async ({ query }, ctx) => ctx.respond({ query, results: [] }),
    })
  },
})
```

```python Python
import os

from solvapay import create_solvapay
from solvapay.mcp import ResponseContext, create_solvapay_mcp_server
from solvapay.mcp.response import ResponseResult

mcp = create_solvapay_mcp_server(
    solvapay=create_solvapay(),
    public_base_url=os.environ["PUBLIC_BASE_URL"],  # where MCP clients reach this server
    name="research",
)


async def quote(symbol: str, ctx: ResponseContext) -> ResponseResult:
    """Latest price for a symbol."""
    return ctx.respond({"symbol": symbol, "price": 101.5})


mcp.register_payable("quote", handler=quote, description="Latest price for a symbol.")

app = mcp.http_app()  # the MCP endpoint is /mcp
```

```go Go
package main

import (
	"context"
	"encoding/json"
	"log"
	"net/http"
	"os"

	solvapay "github.com/solvapay/solvapay-sdk/go"
)

func main() {
	client, err := solvapay.NewClient(solvapay.Config{}) // SOLVAPAY_SECRET_KEY
	if err != nil {
		log.Fatal(err)
	}
	server, err := solvapay.NewMCPServer(solvapay.MCPServerOptions{
		Client:        client, // the product comes from SOLVAPAY_PRODUCT_REF
		PublicBaseURL: os.Getenv("PUBLIC_BASE_URL"),
		Name:          "research",
	})
	if err != nil {
		log.Fatal(err)
	}

	err = server.RegisterPayable("get_quote", solvapay.PaidTool{
		Title:       "Get quote",
		Description: "Latest price for a ticker symbol.",
		InputSchema: map[string]any{
			"type":       "object",
			"properties": map[string]any{"symbol": map[string]any{"type": "string"}},
			"required":   []any{"symbol"},
		},
		Handler: func(ctx context.Context, args json.RawMessage, rc *solvapay.ResponseContext) (*solvapay.ToolResponse, error) {
			var in struct {
				Symbol string `json:"symbol"`
			}
			if err := json.Unmarshal(args, &in); err != nil {
				return nil, err
			}
			return rc.Respond(map[string]any{"symbol": in.Symbol, "price": 101.5}, solvapay.RespondOptions{}), nil
		},
	})
	if err != nil {
		log.Fatal(err)
	}

	log.Fatal(http.ListenAndServe(":3031", server.Handler())) // MCP endpoint: /mcp
}
```

## Prerequisites

- A SolvaPay account
- A product and at least one plan in the SolvaPay Console
- An MCP server project in TypeScript, Python, or Go

For an otherwise-free tool with a per-customer cap, register it as a free tool (`registerFree`,
`register_free`, `RegisterFree`). See [Add a free allowance to MCP tools](/sdks/typescript/guides/free-allowance).
